# 图的遍历

> 关于图的遍历，有两种形式：dfs，bfs，即深度优先遍历和广度优先遍历

## DFS

先介绍DFS的基本思路和流程

![image-20260222195005991](/images/image-20260222195005991.png)

其实本质就是一头往前扎，直到走到死路就往回选择另一条路接着走。

### 邻接矩阵

回忆一下邻接矩阵的存储方式,我们用 G[u] [v]=w 表示点 u 到点 v 这条边的权值为 w ,
现在我们知道点 Src ,想要知道它和哪一个点存在边,我们可以枚举每一个 G[Src] [i] 其中 i 是要枚举的点.然后根据 G[Src] [i] 的值来判断点 Src 到点 i 之间是否存在一条边.

这里如果没有非正数权值就可以把 G 的所有元素初始化为0 ,这样对于任意 G[i] [j] 如果 G[i] [j]=0 就代表着点 i 到点 j 之间不存在边.

当任意一点u到点v之间有边,就代表着从点u可以一步抵达点v.这时候我们把点v标记到visited数组防止下次重复访问造成死循环,为此我们还需要特别判断这个点是否被visist数组标记. 

那么通过一个`for` 循环就可以实现对G[u] [i]中i的枚举.

我们来看看代码怎么写:

```c++
#include<bits/stdc++.h>
using namespace std;
int n, m;//n为点数，m为边数
int Src;
vector<vector<int> >G;//二维vector,两个'>'之间要有空格
vector<int> visited;//记录每个点是否被访问过
void dfs(int x) {//dfs(x)表示从点x开始进行搜索
    visited[x] = 1;
    cout << "访问节点:" << x << endl; 
    for (int i = 0;i < G[x].size();i++) {//从节点0开始,你可以根据题目自行修改
        if (G[x][i] > 0 && !visited[i]) {//如果点x与点i相连且点i未被访问过
            dfs(i);//从点i开始进行搜索
        }
    }
}
int main() {
    int u, v, w;
    cin >> Src;//输入起点
    cin >> n >> m;//输入点数和边数
    G.resize(n + 1, vector<int>(n + 1, 0));//初始化二维vector
    visited.resize(n + 1, 0);//初始化记录数组，初始时所有点都未被访问过
    for (int i = 0;i < m;i++) {
        cin >> u >> v >> w;
        G[u][v] = w;//u到v的边权为w
        G[v][u] = w;//如果无向图
    }
    dfs(Src);
    return 0;
}
```

就好比是这张图，我们走出的路径为 1-&gt;4-&gt;0-&gt;2-&gt;3-&gt;5

#### 分析

时间复杂度O(n^2)

空间复杂度O(n^2)

两个复杂度均较大，容易被卡，其实也说明了邻接矩阵的劣势！！

### 邻接表

邻接表采用结构体的方式，这里由于使用vector套数组，简便好用，即在for循环递归时，只需要满足两点之间存在边即可，然后DFS部分同样。

#### 代码

```c++
#include<bits/stdc++.h>
using namespace std;
const int MAXN = 100005;//数组大小
struct node {
    int v;
    int w;
};
vector<node>G[MAXN];
bool visited[MAXN];//标记是否访问过的数组
int n, m;//n个点m条边
int Src;//起点
void dfs(int u) {//dfs(u)表示从点u开始进行搜索
    visited[u] = 1;//标记已访问
    cout << "访问节点:" << u << endl;
    for (int i = 0;i < G[u].size();i++) {//遍历u的所有邻接点
        if (!visited[G[u][i].v])//如果v未访问过
            dfs(G[u][i].v);//递归访问
    }
}
int main() {
    int u, v, w;//起点,终点,权值
    cin >> Src;//输入起点
    cin >> n >> m;//输入点数和边数
    for (int i = 1;i <= m;i++) {
        cin >> u >> v >> w;
        G[u].push_back((node) { v, w });
        G[v].push_back((node) { u, w });//无向图
    }
    dfs(Src);//从起点开始遍历
    return 0;
}
```

#### 分析

时间复杂度：O(m+n)，其中n为顶点数，m为边数

空间复杂度：O(n)

### 链式前向星

存储结构：定义一个数组head和edge，head[i]存储以节点i为起点的所有边的起始位置，edge[i]存储第i条边的信息。

链式前向星因为是链式结构，所以不需要像邻接矩阵那样枚举点，因为有nxt存储下一条边的编号，当nxt为-1时，就代表所有边已经访问过了。

那么对于这个`for` 循环,我们需要改进一下,将i初始化为head[u],u是`DFS(u)`正在处理的点,让i一直等于nxt[i],这样直到i=-1就可以终止循环了

#### 代码

```c++
#include<bits/stdc++.h>
using namespace std;
const int MAXN = 100000;
struct Edge {
    int to, w, nxt;//目标点，权值，下一条边的编号
}edge[MAXN];
int head[MAXN], cnt;//head[i]表示i号点的第一条边，cnt表示边的数量的一个计数器
bool visited[MAXN];//标记数组，表示是否访问过
int n, m;//n个点，m条边
int Src;//起点
void add(int u, int v, int w) {
    edge[cnt].to = v;
    edge[cnt].w = w;
    edge[cnt].nxt = head[u];//将这条边添加到链表的前面
    head[u] = cnt++;//更新第一条边的编号
}
void dfs(int u) {
    visited[u] = 1;//标记为已访问
    cout << "访问节点:" << u << endl;
    for (int i = head[u]; i != -1; i = edge[i].nxt) {//从head[u]开始,寻找所有关于u的邻接点
        int v = edge[i].to;//获取下一个点
        if (!visited[v]) {//如果未标记
            dfs(v);
        }
    }
}
int main() {
    memset(head, -1, sizeof(head));//初始化head数组为-1
    int u, v, w;
    cin >> Src >> n >> m;
    for (int i = 0; i < m; i++) {
        cin >> u >> v >> w;
        add(u, v, w);
        add(v, u, w);//无向图
    }
    dfs(Src);
    return 0;
}
```

> 这是链式前向星的通用结构，对于add部分，可以理解为通过已知顶点去找他的邻居对应的边，再通过nxt去访问这条边上相关信息

<img src="/images/image-20260222233631638.png" alt="image-20260222233631638" style="zoom:80%;" />

同样是这个样例，它输出的结果为：1-&gt;5-&gt;4-&gt;2-&gt;3-&gt;0

#### 分析

时间复杂度：O(n+m)，n表示图中顶点数，m表示图中边数

空间复杂度：O(n+m)

## BFS

同样，先介绍bfs相关的思路和流程



![image-20260223002311565](/images/image-20260223002311565.png)

> 总结说就是从一个节点开始，往深处遍历，并且不断延伸节点以进行搜索

### 邻接矩阵

由于对于bfs思路很清楚，这里直接上代码！

#### 代码

```c++
#include<bits/stdc++.h>
using namespace std;
int n, m;//n个点，m条边
vector<vector<int> >G;
vector<bool>visited;
int Src;//起点
void bfs(int x) {
    queue<int>q;//队列
    q.push(x);//将起点加入队列
    visited[x] = 1;//标记起点已访问
    while (!q.empty()) {//只有队列不为空才循环
        int u = q.front();//取出队首元素
        q.pop();//将队首元素移出队列
        cout << "访问节点:" << u << endl;
        for (int i = 0;i < n;i++) {//将所有的相邻节点放入队列
            if (G[u][i] > 0 && !visited[i]) {//枚举合法的相邻节点
                visited[i] = 1;//标记该节点已访问
                q.push(i);//将相邻节点加入队列
            }
        }
    }
}
int main() {
    int u, v, w;
    cin >> Src;//输入起点
    cin >> n >> m;//输入点数和边数
    G.resize(n, vector<int>(n, 0));//初始化二维vector
    visited.resize(n, 0);
    for (int i = 0;i < m;i++) {
        cin >> u >> v >> w;
        G[u][v] = w;
        G[v][u] = w;//无向图
    }
    bfs(Src);
    return 0;
}
```

### 链式前向星

链式前向星的BFS实现与邻接表类似，核心差异在于邻接点的遍历方式。我们通过`head[u]`获取首条边的编号，再通过`nxt`指针遍历所有邻接边。

#### 代码

```c++
 #include<bits/stdc++.h>
 using namespace std;
 const int MAXN = 100000;
 struct Edge {
     int to, w, nxt; // 目标点，权值，下一条边的编号
 }edge[MAXN];
 int head[MAXN], cnt; // head初始化为-1，cnt记录边数
 bool visited[MAXN];  // 访问标记数组
 int n, m;            // 顶点数和边数
 int Src;             // 起点
 void add(int u, int v, int w) {
     edge[cnt].to = v;
     edge[cnt].w = w;
     edge[cnt].nxt = head[u]; // 新边插入链表头部
     head[u] = cnt++;         // 更新头指针
 }
 void bfs(int start) {
     queue<int> q;
     visited[start] = true; // 标记起点已访问
     q.push(start);         // 起点入队
     while (!q.empty()) {
         int u = q.front(); // 取出队首节点
         q.pop();
         cout << "访问节点:" << u << endl;
         // 遍历u的所有邻接边
         for (int i = head[u]; i != -1; i = edge[i].nxt) {
             int v = edge[i].to;
             if (!visited[v]) {
                 visited[v] = true; // 标记访问
                 q.push(v);         // 邻接点入队
             }
         }
     }
 }
 int main() {
     memset(head, -1, sizeof(head)); // 初始化head数组
     int u, v, w;
     cin >> Src >> n >> m;
     // 添加边（无向图需添加两次）
     for (int i = 0; i < m; i++) {
         cin >> u >> v >> w;
         add(u, v, w);
         add(v, u, w); // 无向图反向边
     }
     bfs(Src);
     return 0;
 }
```

Ps.注意，这里的head数组初始化为-1是因为cnt开始为0，处理的时候第一次是cnt++，也就是开始为0，head不能赋值0，否则会冲突

不然就得换种写法，如下

```c++
#include<iostream>
#include<cstdio>
#include<cstring>
#include<queue>
using namespace std;
typedef long long ll;
const int MAXN=1e5+5;
struct Edge
{
    int v,w,nxt;
} e[MAXN<<1];
int head[MAXN],tot,n,m;
ll dis[MAXN];
bool vis[MAXN];
inline void add(int u,int v,int w)
{
    e[++tot].v=v;
    e[tot].w=w;
    e[tot].nxt=head[u];
    head[u]=tot;
}
struct Compare{bool operator()(int u,int v){return dis[u]>dis[v];}};
int ans[MAXN];
void dij()
{
    priority_queue<int,vector<int>,Compare> q;
    q.push(1);
    dis[1]=0;
    while(!q.empty())
    {
        int u=q.top();
        q.pop();
        if(vis[u]) continue;
        vis[u]=1;
        for(int i=head[u];i;i=e[i].nxt)
        {
            int v=e[i].v,w=e[i].w;
            if(dis[u]+w<dis[v])
            {
                dis[v]=dis[u]+w;
                ans[v]=u;
                q.push(v);
            }
        }
    }
}
int out[MAXN],otot;
int main()
{
    memset(dis,0x3f,sizeof dis);
    scanf("%d%d",&n,&m);
    for(int i=1;i<=m;i++)
    {
        int u,v,w;
        scanf("%d%d%d",&u,&v,&w);
        add(u,v,w);
        add(v,u,w);
    }
    dij();
    for(int i=n;i;i=ans[i])
    {
        if(i==1)
        {
            printf("1 ");
            for(int j=otot;j>=1;j--) printf("%d ",out[j]);
            return 0;
        }
        out[++otot]=i;
    }
    printf("-1");
    return 0;
}
```

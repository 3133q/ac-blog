# 线段树进阶

> 对于常规线段树，可以查询区间和，以及进行单点修改，但如果我要求的是区间修改呢？？

## 回顾

对于常规线段树，先复习复习基本思路。

首先，肯定是把数先建好，对于tree数组的存储空间，可以这么想，最终建成的树的结点数其实就是元素的数量乘2，但是我们考虑到，树不一定是满二叉树，也就是说单纯定义2 * N的大小的话，那么剩下没有存值的结点没有给他们分配相应的存储空间，这样就很容易炸，所以正确应该定义4 * N，对于建树过程的话，递归就好了

其次，pushup操作，也就是累加求和，把和不断往上递增

然后的话，query和change操作还是建议写mid（感觉更保险哈哈哈），通过mid与l，r比较，明确应该寻找左子树还是右子树，然后返回个sum值就行啦

下面来看看代码：

```c++
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n,m,p,q,t;
const int N=1e7+2;
vector<int>a;
struct node{
	int sum,l,r;
}tree[4*N];
inline void pushup(int x){
	tree[x].sum=tree[2*x].sum+tree[2*x+1].sum;
}
void build(int x,int l,int r){
	tree[x].l=l,tree[x].r=r;
	if(l==r){
		tree[x].sum=a[l];
		return ;
	}
	int mid=l+((r-l)>>1);
	build(2*x,l,mid);
	build(2*x+1,mid+1,r);
	pushup(x);
}
int query(int x,int l,int r){
	if(tree[x].l>=l&&tree[x].r<=r) return tree[x].sum;
	int mid=tree[x].l+((tree[x].r-tree[x].l)>>1);
	int s=0;
	if(l<=mid) s+=query(2*x,l,r);
	if(r>mid) s+=query(2*x+1,l,r);
	return s;
}
void change(int now,int x,int k){
	if(tree[now].l==tree[now].r){
		tree[now].sum+=k;
		return ;
	}
	int mid=tree[now].l+((tree[now].r-tree[now].l)>>1);
	if(x<=mid) change(2*now,x,k);
	else change(2*now+1,x,k);
	pushup(now);
}
int main(){
	std::cin>>n>>m;
	a.resize(n+1);
	for(int i=1;i<=n;i++) std::cin>>a[i];
	build(1,1,n);
	while(m--){
		std::cin>>p>>q>>t;
		if(p==1) change(1,q,t);
		else std::cout<<query(1,q,t)<<'\n';
	}
	return 0;
}
```

### 注意

对于线段树来说（基础版），他可以支持单点修改和区间查询，这里有很多变式，有些变式可以不需要建树的，因为建树的目的在于把n个结点分别放进线段树中，然后在进行区间查询和单点修改时直接对当前的数值进行操作，但如果对于当前无需初始化每个结点的值呢？这时，其实就可以跳过建树的步骤，因为写了的话也是把sum赋值为0，所以可以不写，以这道题为例：

### 例题

[P3353 在你窗外闪耀的星星 - 洛谷](https://www.luogu.com.cn/problem/P3353)

#### 分析

相当于在一个数轴上，给出了几个点的坐标，这里如果想着建树的话，会发现很麻烦，也不好建，索性干脆不建了，在后面操作的时候直接转mid就OK了，因为n给出的范围是100000，不算非常大，所以可以直接枚举，枚举1到100000，作为数轴所有的点，然后在单点修改时，直接1~100000作为l，r，去查要的结点在左边还是右边，二分的思想，找到以后直接加上值就行，而对于区间查询，给出目标区间和原始区间1-100000，然后一步步去定位，定位到以后把和输出来即可！

#### 代码

```c++
#include<bits/stdc++.h>
using namespace std;
using ll=long long;
const int N=1e5+2;
struct node{
	ll l,r;
	ll sum;
}tree[N*4];
void pushup(ll x){
	tree[x].sum=tree[x<<1].sum+tree[x<<1|1].sum;
}
void build(ll x,ll l,ll r){
	tree[x].l=l,tree[x].r=r;
	if(l==r){
		tree[x].sum=0;
		return ;
	}
	ll mid=tree[x].l+tree[x].r>>1;
	build(x<<1,l,mid);
	build(x<<1|1,mid+1,r);
	pushup(x);
}
ll query(ll x,ll l,ll r,ll L,ll R){
	if(L>=l&&R<=r) return tree[x].sum;
	ll sum=0;
	ll mid=L+R>>1;
	if(l<=mid) sum+=query(x<<1,l,r,L,mid);
	if(r>mid) sum+=query(x<<1|1,l,r,mid+1,R);
	return sum;
}
void change(ll now,ll p,ll k,ll l,ll r){
	if(l==r) {
		tree[now].sum+=k;
		return ;
	}
	ll mid=l+r>>1;
	if(p<=mid) change(now<<1,p,k,l,mid);
	else change(now<<1|1,p,k,mid+1,r);
	pushup(now);
}
int main(){
	ll n,w,x[N],y[N];
	std::cin>>n>>w;
	for(int i=1;i<=n;i++) std::cin>>x[i]>>y[i];
	if(w==0){
		std::cout<<0;
		return 0;
	}
	//build(1,1,n);
	for(int i=1;i<=n;i++) change(1,x[i],y[i],1,100000);
	ll maxx=INT_MIN;
	ll i,j,s=0;
	for(i=1;i<=100000;i++){
		if(i-w+1<0) continue;
		maxx=max(maxx,query(1,i-w+1,i,1,100000));
	}
	std::cout<<maxx;
	return 0;
}
```

> 滑窗部分直接这么写就OK了，没必要断开来。。。

## 进阶

> 想必你已经看出来了吧，对于基础版本的线段树，一直在强调单点修改，单点！那如果我要区间修改呢？那你不就没招了哈哈哈

进阶--区间修改+区间查询

### 引入

其实我们想从单点修改到覆盖到整个区间，无非就是得有个标记，记录一下当前改变了多少，然后往下传，把整个区间都覆盖到，这样最后计算的时候，就可以涵盖了

用到的东西我们称为--懒标记

### 方法

先给懒标记add标记为0，只有在change操作时，它才会发生改变，然后我们来讨论一下sum该怎么算

因为我们事先把tree[x].l，tree[x].r已经计算好，即为对应元素在集合中的位置，这样的话，算sum的时候，就可以直接利用l，r来获取当前节点对应的元素数量有多少，就是r-l+1，然后把数量乘上add标记即可，最后把当前节点的懒标记加上旧的，随后旧的清0，因为它已经传递完成了，没用途了。。

具体定义为pushdown操作，一步步往下传标记

### 代码

```c++
#include<bits/stdc++.h>
#define ll long long
#define N 100005 
using namespace std;
ll n,m,a[N];
struct node{
	ll l,r;
	ll sum;
	ll add;
}tree[4*N];
inline void pushup(ll x){
	tree[x].sum=tree[x<<1].sum+tree[x<<1|1].sum;
}
void build(ll x,ll l,ll r){
	tree[x]={l,r,a[l],0};
	if(l==r) return ;
	ll mid=l+r>>1;
	build(2*x,l,mid);
	build(2*x+1,mid+1,r);
	pushup(x);
}
void pushdown(ll p){
	if(tree[p].add){
		tree[p<<1].sum+=tree[p].add*(tree[p<<1].r-tree[p<<1].l+1);
		tree[p<<1|1].sum+=tree[p].add*(tree[p<<1|1].r-tree[p<<1|1].l+1);
		tree[p<<1].add+=tree[p].add;
		tree[p<<1|1].add+=tree[p].add;
		tree[p].add=0;
	}
}
ll query(ll x,ll l,ll r){
	if(tree[x].l>=l&&tree[x].r<=r) return tree[x].sum;
	ll mid=tree[x].l+tree[x].r>>1;
	pushdown(x);
	ll sum=0;
	if(l<=mid) sum+=query(x<<1,l,r);
	if(r>mid) sum+=query(x<<1|1,l,r);
	return sum;
}
void change(ll now,ll l,ll r,ll k){
	if(tree[now].l>=l&&tree[now].r<=r){
		tree[now].sum+=(tree[now].r-tree[now].l+1)*k;
		tree[now].add+=k;
		return ;
	}
	ll mid=tree[now].l+tree[now].r>>1;
	pushdown(now);
	if(l<=mid) change(now<<1,l,r,k);
	if(r>mid) change(now<<1|1,l,r,k);
	pushup(now);
}
int main(){
	std::cin>>n>>m;
	for(int i=1;i<=n;i++) std::cin>>a[i];
	build(1,1,n);
	ll p,x,y,k;
	for(int i=1;i<=m;i++){
		std::cin>>p;
		if(p==1){
			std::cin>>x>>y>>k;
			change(1,x,y,k);
		}
		else {
			std::cin>>x>>y;
			std::cout<<query(1,x,y)<<'\n';
		}
	}
	return 0;
}
```

## 再进阶

既然区间修改可以加，那如果我再加上个乘呢？？？

那自然要用到两个懒标记。。。

### 举例

让我们看看实行加法和乘法后区间和的变化
$$
s=s+v_1\\
s=(s+v_1)\times u_1=s\times u_1+v_1\times u_1\\
s=s\times u_1+v_1\times u_1+v_2\\
s=(s\times u_1+v_1\times u_1+v_2)\times u_2=s\times u_1\times u_2+v_1\times u_1\times u_2+v_2\times u_2
$$
进行多次区间修改后，最终区间和的变化可以描述为：区间中每个元素先乘以某数值，每个元素再加上某数值。注意要**先乘后加**。

假设两个标记为mult，add，那么sum=mult * sum+(r-l+1) * add

对于add函数

### 函数解释

[l,r]每个元素乘以v

区间和变化：
$$
sum=sum\times v
$$
标记变化：
$$
sum=(mult\cdot sum+(r-l+1)\cdot add)\cdot v=mult\cdot v\cdot sum+(r-l+1)\cdot add\cdot v
$$
因此：
$$
mult=mult\cdot v \quad add=add\cdot v
$$
[l,r]每个元素加上v

区间和变化：
$$
sum=sum+v
$$
标记变化：
$$
sum=mult\cdot sum+(r-l+1)\cdot add+(r-l+1)\cdot v=mult\cdot sum+(r-l+1)\cdot (add+v)
$$
因此：
$$
add=add+v
$$
利用这个函数传标记即可，其余部分一致

#### 注意

要先乘后加！！！因为最后的sum结果是sum=sum * mult+(r-l+1) * add，必须先执行乘法操作，然后加法，不然的话就会出问题！！！

### 例题

[P3373 【模板】线段树 2 - 洛谷](https://www.luogu.com.cn/problem/P3373)

### 代码

```c++
#include<bits/stdc++.h>
using namespace std;
using ll=long long;
const int N=1e5+2;
struct node{
	ll l,r;
	ll add;
	ll mult;
	ll sum;
}tree[N*4];
ll n,m,mod,a[N],p,x,y,k;
void pushup(ll x){
	tree[x].sum=(tree[x<<1].sum+tree[x<<1|1].sum)%mod;
}
void build(ll x,ll l,ll r){
	tree[x]={l,r,0,1,a[l]};
	if(l==r) return ;
	ll mid=l+r>>1;
	build(x<<1,l,mid);
	build(x<<1|1,mid+1,r);
	pushup(x);
}
void add(ll type,ll x,ll v){
	if(type==2){
		tree[x].add=(tree[x].add+v)%mod;
		tree[x].sum=(tree[x].sum+(tree[x].r-tree[x].l+1)*v)%mod;
	}
	if(type==1){
		tree[x].add=tree[x].add*v%mod;
		tree[x].mult=tree[x].mult*v%mod;
		tree[x].sum=tree[x].sum*v%mod;
	}
}
void pushdown(ll x){
	if(tree[x].add==0&&tree[x].mult==1) return ;
	add(1,x<<1,tree[x].mult);
	add(1,x<<1|1,tree[x].mult);
	add(2,x<<1,tree[x].add);
	add(2,x<<1|1,tree[x].add);
	tree[x].add=0,tree[x].mult=1;
}
void change(ll type,ll now,ll l,ll r,ll k){
	if(tree[now].l>=l&&tree[now].r<=r){
		add(type,now,k);
		return ;
	}
	pushdown(now);
	ll mid=tree[now].l+tree[now].r>>1;
	if(l<=mid) change(type,2*now,l,r,k);
	if(r>mid) change(type,2*now+1,l,r,k);
	pushup(now);
}
ll query(ll x,ll l,ll r){
	if(tree[x].l>=l&&tree[x].r<=r) return tree[x].sum;
	pushdown(x);
	ll sum=0;
	ll mid=tree[x].l+tree[x].r>>1;
	if(l<=mid) sum+=query(2*x,l,r);
	if(r>mid) sum+=query(2*x+1,l,r);
	return sum%mod;
}
int main(){
	std::cin>>n>>m>>mod;
	for(int i=1;i<=n;i++) std::cin>>a[i];
	build(1,1,n);
	for(int i=1;i<=m;i++){
		std::cin>>p;
		if(p==1){
			std::cin>>x>>y>>k;
			change(1,1,x,y,k);
		}
		if(p==2){
			std::cin>>x>>y>>k;
			change(2,1,x,y,k);
		}
		else if(p==3) {
			std::cin>>x>>y;
			std::cout<<query(1,x,y)<<'\n';
		}
	}
	return 0;
}
```

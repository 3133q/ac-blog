```c++
#include<iostream>
#include<string>
using namespace std;
const int N=1e6+2;
int pmt[N];
void get_pmt(const string& s){
	pmt[0]=0;
	int l=s.length();
	for(int i=1,j=0;i<l;i++){
		while(j&&s[i]!=s[j]) j=pmt[j-1];
		if(s[i]==s[j]) j++;
		pmt[i]=j;
	}
}
void kmp(const string& s1,const string& s2){
	int l1=s1.length(),l2=s2.length();
	for(int i=0,j=0;i<l1;i++){
		while(j&&s1[i]!=s2[j]) j=pmt[j-1];
		if(s1[i]==s2[j]) j++;
		if(j==l2){
			std::cout<<i-j+2<<'\n';
			j=pmt[j-1];
		}
	}
}
int main(){
	string s1,s2;
	std::cin>>s1>>s2;
	get_pmt(s2);
	kmp(s1,s2);
	for(int i=0;i<s2.length();i++){
		std::cout<<pmt[i]<<" ";
	}
	return 0;
}
```

## 2.差分

**一维差分----**

d[i]=a[i]-a[i-1]

C++ 标准库中实现了差分函数 [`std::adjacent_difference`](https://zh.cppreference.com/w/cpp/algorithm/adjacent_difference)，定义于头文件 `<numeric>` 中。

假设要将序列 {𝑎𝑖}![\{a_i\}](data:image/gif;base64,R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7) 在区间 [𝑙,𝑟]![[l,r]](data:image/gif;base64,R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7) 中的每个数都加上一个 𝑣![v](data:image/gif;base64,R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7)。可以在它的差分序列 {d𝑖}![\{D_i\}](data:image/gif;base64,R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7) 上做如下操作：

d[l]+=v       d[r+1]-=v

对于  1 3 2 4 5

进行差分    1 2 -1 2 1

然后前缀和  1 3 2 4 5    发现回到了最先开始的状态，即前缀和，差分互逆

应用：对一个区间进行加法/减法运算时，可以利用差分算法

eg.   对[l,r] 之间的每一个数加上c

实现  先化为差分数组，然后a[l]+=c，a[r+1]-=c

验证    1 3 2 4 5        对[2,4] 进行加2操作

差分    1 2 -1 2 1 0          ----&gt;          1 4 -1 2 -1 0    （有时可以多加一项0，避免溢出）

前缀和   1 5 4 6 5        实现操作

**二维差分----**

d[i] [j]=a[i] [j]-a[i-1] [j]-a[i] [j-1]+a[i-1] [j-1]

二维差分信息常用于维护二维数组的多次矩形加。例如，要对左上角为 (𝑥1,𝑦1)![(data:image/gif;base64,R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7)](data:image/gif;base64,R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7)、右下角为 (𝑥2,𝑦2)![(data:image/gif;base64,R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7)](data:image/gif;base64,R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7) 的矩阵中的每个数字都加上 𝑣![v](data:image/gif;base64,R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7)，可以对它的差分数组 {𝐷𝑖,𝑗}![\{D_{i,j}\}](data:image/gif;base64,R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7) 做如下操作：

d[x1] [y1]+=v        d[x1] [y2+1]-=v          d[x2+1] [y1]-=v        d[x2+1] [y2+1] +=v

### 矩阵快速幂

实则就是将快速幂函数内的base换为一个单位矩阵，将数与数的乘积变化为矩阵和矩阵之间的乘积，以斐波那契数列为例

这么想   数列满足F(n)=F(n-1)+F(n-2)

假设矩阵[F(n),F(n-1)]=[F(n-1),F(n-2)]* ![](/images/32.png)

容易解得 a=1,b=1,c=1,d=0;

<img src="/images/33.png" style="zoom:75%;" />

这里使用二维数组d[2] [2]定义矩阵，并在struct中重载乘法运算；快速幂算法采取非递归快速幂。

代码实现--

```c++
#include<bits/stdc++.h>
using namespace std;

typedef long long lint;

lint m, n;

struct Matrix {
    lint d[2][2];

    Matrix() {
        memset(d, 0, sizeof(d));
    }

    Matrix operator* (const Matrix& x) {
        Matrix ans;
        for (int i = 0; i <= 1; ++i) {
            for (int j = 0; j <= 1; ++j) {
                for (int k = 0; k <= 1; ++k) {
                    ans.d[i][j] += (d[i][k] * x.d[k][j]) % m;
                }
                ans.d[i][j] = (((ans.d[i][j] % m) + m) % m);
            }
        }
        return ans;
    }
};

Matrix fpow(Matrix a, lint n) {
    Matrix ans;
    ans.d[0][0] = ans.d[1][1] = 1;//初始化
    while (n) {
        if (n & 1) {
            ans = ans * a;
        }
        a = a * a;
        n >>= 1;
    }
    return ans;
}

int main() {
    cin >> m >> n;
    if (n == 1) {
        cout << 1 << endl;
        return 0;
    } else if (n == 2) {
        cout << 1 << endl;
        return 0;
    }
    Matrix a;
    a.d[0][0] = 1;
    a.d[0][1] = 1;

    Matrix c;
    c.d[0][0] = 1;
    c.d[0][1] = 1;
    c.d[1][0] = 1;
    c = fpow(c, n - 2);
    a = a * c;
    cout << a.d[0][0] << endl;

    return 0;
}
```



## 6.匈牙利算法

```c++
#include <bits/stdc++.h>
using namespace std;

const int N=5e2+5,M=5e4+5;
int e[M],ne[M],h[N],idx;
int mate[N],st[N];

void add(int a,int b) {
    e[idx]=b,ne[idx]=h[a],h[a]=idx++;
}

bool HA(int u) { //Hungarian Algorithm
    for(int i=h[u]; i!=-1; i=ne[i]) {
        int j=e[i];
        if(!st[j]) {
            st[j]=true;
            if(!mate[j] || HA(mate[j])) {
                mate[j]=u;
                return true;
            }
        }
    }
    return false;
}

int main() {
    memset(h,-1,sizeof h);
    int n,m,e;
    cin>>n>>m>>e;
    while(e--) {
        int u,v;
        cin>>u>>v;
        add(u,v);
    }

    int cnt=0;
    for(int i=1; i<=n; i++) {
        memset(st,false,sizeof st);
        if(HA(i)) cnt++;
    }
    cout<<cnt<<endl;

    return 0;
}
```

## 7.并查集

```c++
#include<bits/stdc++.h>
using namespace std;
const int N=1e4+2;
int rank1[N],fa[N],n;
inline void init(){
	for(int i=1;i<=n;i++){
		rank1[i]=1;
		fa[i]=i;
	}
}
int find(int i){
	return fa[i]==i?fa[i]:(fa[i]=find(fa[i]));
}
void merge(int i,int j){
	int x=find(i),y=find(j);
	if(rank1[x]<=rank1[y]) fa[x]=y;
	else fa[y]=x;
	if(rank1[x]==rank1[y]&&x!=y) rank1[y]++;
	return ;
}
int main(){
	int m,q,x,y,x1,y1; 
	std::cin>>n>>m>>q;
	init();
	for(int i=1;i<=m;i++){
		std::cin>>x>>y;
		merge(x,y);
	}
	for(int i=1;i<=q;i++){
		std::cin>>x1>>y1;
		if(find(x1)==find(y1)) std::cout<<"Yes"<<'\n';
		else std::cout<<"No"<<'\n';
	}
	return 0;
}
```

## 8.拓扑排序

```c++
#include<iostream>
#include<cstdio>
#include<cstring>
#include<algorithm>
#include<climits>
#include<queue>
#include<vector>
using namespace std;
const int N=1001;
int main(){
	int n,j,total=0;
	std::cin>>n;
	vector<int>inq(n+1);
	vector<int>topo(n+1);
	vector<int>edge[N];
	for(int i=1;i<=n;i++){
		while(std::cin>>j&&j!=0){
			edge[i].push_back(j);
			inq[j]++;
		}
	}
	queue<int>q;
	for(int i=1;i<=n;i++){
		if(inq[i]==0) q.push(i);
	}
	while(!q.empty()){
		int u=q.front();
		q.pop();
		topo[total++]=u;
		for(auto e:edge[u]){
			inq[e]--;
			if(inq[e]==0) q.push(e);
		}
	}
	for(int i=0;i<total;i++) std::cout<<topo[i]<<" ";
	std::cout<<'\n';
	return 0;
}
```

最后给出判断是否存在环的方法

使用拓扑排序可以判断一个无向图中是否存在环，具体步骤如下：

1. 求出图中所有结点的度。
2. 将所有度 &lt;= 1 的结点入队。（独立结点的度为 0）
3. 当队列不空时，弹出队首元素，把与队首元素相邻节点的度减一。如果相邻节点的度变为一，则将相邻结点入队。
4. 循环结束时判断已经访问的结点数是否等于 n。等于 n 说明全部结点都被访问过，无环；反之，则有环。

有向图的话，把入度改为等于0的时候入队就行了

### 代码（有向）

```c++

		inq[y]++;
	}

			ans++; 
			inq[e]--;
			if(inq[e]==0) q.push(e);
		}
	}
	if(ans==n) std::cout<<"NO";
	else std::cout<<"YES";
	return 0;
}
```

### Floyd

```c++
#include <bits/stdc++.h>
using namespace std;
const int INF = 1e6; // 路口之间的初始距离为无穷大，相当于断开
const int NUM = 105;
int graph[NUM][NUM]; // 邻接矩阵存图
int n, m;
void floyd()
{
    int s = 1;                   // 起点
    for (int k = 1; k <= n; k++) // floyd()的三重循环
        for (int i = 1; i <= n; i++)
            if (graph[i][k] != INF) // 小优化
                for (int j = 1; j <= n; j++)
                    if (graph[i][j] > graph[i][k] + graph[k][j])
                        graph[i][j] = graph[i][k] + graph[k][j]; // 用min()也可，但比较慢可能超时
    printf("%d\n", graph[s][n]);
}
int main()
{
    while (~scanf("%d%d", &n, &m))
    {
        if (n == 0 && m == 0)
            return 0;
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= n; j++)
                graph[i][j] = INF; // 邻接矩阵初始化，任意两点间初始距离为无穷大
        while (m--)
        {
            int a, b, c;
            scanf("%d%d%d", &a, &b, &c);
            graph[a][b] = graph[b][a] = c; // 邻接矩阵存图
        }
        floyd();
    }
    return 0;
}
```

### SPFA

```c++
#include<iostream>
#include<cstdio>
#include<cstring>
#include<climits>
#include<algorithm>
#include<queue>
using namespace std;
using ll=long long;
const int N=1e4+2;
struct node{
	ll v,w,next;
}edge[N];
ll cnt,dis[N],head[N],inq[N],neg[N],n;
void add(ll u,ll v,ll w){
	edge[++cnt].v=v;
	edge[cnt].w=w;
	edge[cnt].next=head[u];
	head[u]=cnt;
}
bool spfa(){
	dis[1]=0;
	queue<ll>q;
	q.push(1);
	inq[1]=1;
	while(!q.empty()){
		ll u=q.front();
		q.pop();
		inq[u]=0;
		for(int i=head[u];i;i=edge[i].next){
			ll v=edge[i].v,w=edge[i].w;
			if(dis[v]>dis[u]+w){
				dis[v]=dis[u]+w;
				neg[v]=neg[u]+1;
				if(neg[v]>=n) return true;
				if(!inq[v]) {
					inq[v]=1;
					q.push(v);
				}
			}
		}
	}
	return false;
}
void solve(){
	ll m,u,v,w;
	memset(dis,0x3f,sizeof(dis));
	memset(neg,0,sizeof(neg));
	memset(inq,0,sizeof(inq));
	memset(edge,0,sizeof(edge));
	memset(head,0,sizeof(head));
	cnt=0;
	std::cin>>n>>m;
	for(int i=1;i<=m;i++){
		std::cin>>u>>v>>w;
		add(u,v,w);
		if(w>=0) add(v,u,w);
	}
	if(spfa()==1) std::cout<<"YES"<<'\n';
	else std::cout<<"NO"<<'\n';
	return ;
}
int main(){
	int t;
	std::cin>>t;
	for(int i=0;i<t;i++) solve();
	return 0;
}
```

### Dijkstra

```c++
#include<iostream>
#include<cstdio>
#include<cstring>
#include<algorithm>
#include<climits>
#include<queue>
using namespace std;
using ll=long long;
const int N=1e6+2;
struct node{
	ll v,w,next;
}edge[N];
ll head[N],cnt,n,m,s,vis[N],dis[N];
void add(ll u,ll v,ll w){
	edge[++cnt].v=v;
	edge[cnt].w=w;
	edge[cnt].next=head[u];
	head[u]=cnt;
}
void dijkstra(){
	dis[s]=0;
	priority_queue<pair<ll,ll>,vector<pair<ll,ll>>,greater<pair<ll,ll>>>q;
	q.push({0LL,s});
	while(!q.empty()){
		ll u=q.top().second;
		q.pop();
		if(vis[u]) continue;
		vis[u]=1;
		for(int i=head[u];i;i=edge[i].next){
			ll v=edge[i].v,w=edge[i].w;
			if(dis[v]>dis[u]+w){
				dis[v]=dis[u]+w;
				q.push({dis[v],v});
			}
		}
	}
	for(int i=1;i<=n;i++) std::cout<<dis[i]<<" ";
	return ;
}
int main(){
	ll u,v,w;
	memset(dis,0x3f,sizeof(dis));
	std::cin>>n>>m>>s;
	for(int i=1;i<=m;i++){
		std::cin>>u>>v>>w;
		add(u,v,w);
	}
	dijkstra();
}
```



## 10.单调队列

```c++
#include<iostream>

const int N=1e6+2;
ll a[N],q[N],n,k,h,t;
void get_min(){
	h=1,t=0;
	for(int i=1;i<=n;i++){
		if(h<=t&&q[h]<i-k+1) h++;
		while(h<=t&&a[q[t]]>=a[i]) t--;
```

```c++
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n,m,p,q,t;
const int N=1e7+2;
vector<int>a;
struct node{
	int sum,l,r;
}tree[4*N];
inline void pushup(int x){
	tree[x].sum=tree[2*x].sum+tree[2*x+1].sum;
}
void build(int x,int l,int r){
	tree[x].l=l,tree[x].r=r;
	if(l==r){
		tree[x].sum=a[l];
		return ;
	}
	int mid=l+((r-l)>>1);
	build(2*x,l,mid);
	build(2*x+1,mid+1,r);
	pushup(x);
}
int query(int x,int l,int r){
	if(tree[x].l>=l&&tree[x].r<=r) return tree[x].sum;
	int mid=tree[x].l+((tree[x].r-tree[x].l)>>1);
	int s=0;
	if(l<=mid) s+=query(2*x,l,r);
	if(r>mid) s+=query(2*x+1,l,r);
	return s;
}
void change(int now,int x,int k){
	if(tree[now].l==tree[now].r){
		tree[now].sum+=k;
		return ;
	} 
	int mid=tree[now].l+((tree[now].r-tree[now].l)>>1);
	if(x<=mid) change(2*now,x,k);
	else change(2*now+1,x,k);
	pushup(now);
}
int main(){
```

### 进阶

```c++
#include<bits/stdc++.h>
using namespace std;
using ll=long long;
const int N=1e5+2;
struct node{
	ll l,r;
	ll add;
	ll mult;
	ll sum;
}tree[N*4];
ll n,m,mod,a[N],p,x,y,k;
void pushup(ll x){
	tree[x].sum=(tree[x<<1].sum+tree[x<<1|1].sum)%mod;
}
void build(ll x,ll l,ll r){
	tree[x]={l,r,0,1,a[l]};
	if(l==r) return ;
	ll mid=l+r>>1;
	build(x<<1,l,mid);
	build(x<<1|1,mid+1,r);
	pushup(x);
}
void add(ll type,ll x,ll v){
	if(type==2){
		tree[x].add=(tree[x].add+v)%mod;
		tree[x].sum=(tree[x].sum+(tree[x].r-tree[x].l+1)*v)%mod;
	}
	if(type==1){
		tree[x].add=tree[x].add*v%mod;
		tree[x].mult=tree[x].mult*v%mod;
		tree[x].sum=tree[x].sum*v%mod;
	}
}
void pushdown(ll x){
	if(tree[x].add==0&&tree[x].mult==1) return ;
	add(1,x<<1,tree[x].mult);
	add(1,x<<1|1,tree[x].mult);
	add(2,x<<1,tree[x].add);
	add(2,x<<1|1,tree[x].add);
	tree[x].add=0,tree[x].mult=1;
}
void change(ll type,ll now,ll l,ll r,ll k){
	if(tree[now].l>=l&&tree[now].r<=r){
		add(type,now,k);
		return ;
	}
	pushdown(now);
	ll mid=tree[now].l+tree[now].r>>1;
	if(l<=mid) change(type,2*now,l,r,k);
	if(r>mid) change(type,2*now+1,l,r,k);
	pushup(now);
}
ll query(ll x,ll l,ll r){
	if(tree[x].l>=l&&tree[x].r<=r) return tree[x].sum;
	pushdown(x);
	ll sum=0;
	ll mid=tree[x].l+tree[x].r>>1;
	if(l<=mid) sum+=query(2*x,l,r);
	if(r>mid) sum+=query(2*x+1,l,r);
	return sum%mod;
}
int main(){
	std::cin>>n>>m>>mod;
	for(int i=1;i<=n;i++) std::cin>>a[i];
	build(1,1,n);
	for(int i=1;i<=m;i++){
		std::cin>>p;
		if(p==1){
			std::cin>>x>>y>>k;
			change(1,1,x,y,k);
		}
		if(p==2){
			std::cin>>x>>y>>k;
			change(2,1,x,y,k);
		}
		else if(p==3) {
			std::cin>>x>>y;
			std::cout<<query(1,x,y)<<'\n';
		}
	}
	return 0;
}
```

```c++
int A[8], cnt, dp[8][12][2];
int dfs(int pos, int last, bool limit)
{
    int ans = 0;
    if (pos == cnt)
        return 1; // 搜索终点
    if (dp[pos][last][limit] != -1)
        return dp[pos][last][limit];
    for (int v = 0; v <= (limit ? A[pos] : 9); ++v) // 根据是否limit决定循环上界
    {
```

```c++
#include<iostream>

ll qpow(ll a,ll b,ll p){
```

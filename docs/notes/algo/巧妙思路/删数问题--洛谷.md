删数题基本思路---

题目要删去指定数量的数字，使得最后的数字最小，可以采用往后挪位的思路

例如 175438

先第一遍找最小的数 1，然后输出，随后从1的右侧开始继续找最小的数，反复.......

具体代码：

```c++
#include<bits/stdc++.h>
using namespace std;
int n,k,a[257],rest,t=1,minp,cnt=0;
bool flag=0;
string num;
int main(){
    cin>>num>>k;
    n=num.length();
    for(int i=1;i<=n;++i)a[i]=num[i-1]-'0';
    rest=n-k;
    while(cnt<rest){
        minp=t;
        for(int i=t;i<=k+t;++i)if(a[minp]>a[i])minp=i;
        if(a[minp])flag=1;
        if(flag)cout<<a[minp];
        k-=minp-t;
        t=minp+1;
        cnt++;
    }
    if(!flag)cout<<0;
    return 0;
}
```

---

这道题也可以用单调栈的思路去做，因为我们可以想想，既然要让删掉后的数最小，那我们绝对不能允许左边的数字比右边的数字大，那不就直接单调栈去解吗

比如说对于n="1432219" k=3

1.遇到1，栈空，栈[1]

2.遇到4，4&gt;1，栈[1,4]

3.遇到3，3&lt;4，删掉栈顶，k变为2，栈为[1,3]

4.遇到2，2&lt;3，删掉栈顶，k变为1，栈为[1,2]

5.遇到2，2=2，直接加入，栈[1,2,2]

6.遇到1，1&lt;2，删掉栈顶，k=0，栈[1,2,1]

7.因为k=0，之后的数直接加入，即为[1,2,1,9]

代码：

```c++
#include <iostream>
#include <string>

using namespace std;

int main() {
    // 优化输入输出速度（在算法竞赛中很常用）
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    string n;
    int k;
    
    // 读入字符串和需要删除的个数
    if (!(cin >> n >> k)) {
        return 0;
    }

    string res = ""; // res 既是最终结果，也在过程中充当“单调栈”

    // 1. 遍历数字字符串的每一个字符
    for (char digit : n) {
        // 核心贪心逻辑：
        // 当栈不为空 (!res.empty()) 
        // 且还有删除名额 (k > 0) 
        // 且栈顶数字大于当前数字 (res.back() > digit) 时，弹出栈顶！
        while (!res.empty() && k > 0 && res.back() > digit) {
            res.pop_back(); // 删掉左边比较大的数字
            k--;            // 消耗一个删除名额
        }
        // 当前数字入栈
        res.push_back(digit);
    }

    // 2. 边缘情况一：如果原数字是递增的（如 12345），遍历完了 k 还不为 0
    // 那就只能从末尾砍掉剩下的 k 个数字
    while (k > 0 && !res.empty()) {
        res.pop_back();
        k--;
    }

    // 3. 边缘情况二：去除前导 0
    // 找到第一个不是 '0' 的字符的位置
    int start = 0;
    while (start < res.length() && res[start] == '0') {
        start++;
    }
    
    // 截取掉前面的所有 '0'
    res = res.substr(start);

    // 4. 边缘情况三：如果全删光了，或者都是0被去掉了，必须输出 "0"
    if (res.empty()) {
        cout << "0" << "\n";
    } else {
        cout << res << "\n";
    }

    return 0;
}
```

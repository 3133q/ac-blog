巧思-----矩形

[P3217 [HNOI2011\] 数矩形 - 洛谷](https://www.luogu.com.cn/problem/P3217)

给出一系列的点，要找出最大面积的矩形，避免暴力的做法

![img](https://cdn.luogu.com.cn/upload/image_hosting/g734wapy.png)

参考这张图，可以想到这样的巧思：根据矩形对角线长度相等，以此确定是否能构成矩形，可以通过定义结构体来存储连续两个点，并存在mid（两点中值），dist（两点之间长度），由此如果都相等，则这两点确定矩形的两个对角，由此可以直接计算面积

代码如下：

```c++
#include<bits/stdc++.h>
using namespace std;
const int N=2000;
struct node{
	int x1,y1,x2,y2;
	int midx,midy;
	int dist;
}e[N*N];
long long x[N],y[N];
bool cmp(node x,node y){
	if(x.midx==y.midx){
		if(x.midy==y.midy) return x.dist<y.dist;
		return x.midy<y.midy;
	}
	return x.midx<y.midx;
}
inline long long cal(long long x){
	return x*x;
}
int main(){
	long long n,t=0,ans=INT_MIN;
	std::cin>>n;
	for(int i=1;i<=n;i++) std::cin>>x[i]>>y[i];
	for(int i=1;i<n;i++){
		for(int j=i+1;j<=n;j++){
			e[++t].x1=x[i],e[t].y1=y[i];
			e[t].x2=x[j],e[t].y2=y[j];
			e[t].midx=x[i]+x[j];
			e[t].midy=y[i]+y[j];
			e[t].dist=cal(x[i]-x[j])+cal(y[i]-y[j]);
		}
	}
	sort(e+1,e+1+t,cmp);
	int st=0;
	for(int i=1;i<=t;i++){
		if(e[i].midx==e[st].midx&&e[i].midy==e[st].midy){
			for(int j=st;j<=i;j++){
				if(e[i].dist==e[j].dist) 
				  ans=max(ans,(long long)(sqrt(cal(e[i].x1-e[j].x1)+cal(e[i].y1-e[j].y1))*sqrt(cal(e[i].x1-e[j].x2)+cal(e[i].y1-e[j].y2))));
				  //ans=max(ans,(long long)(sqrt(cal(abs(e[i].x1-e[j].x1))+cal(abs(e[i].y1-e[j].y1)))*sqrt(cal(abs(e[i].x1-e[j].x2))+cal(abs(e[i].y1-e[j].y2) ) )));
			}
		}
		else if(e[i].midx!=e[st].midx||e[i].midy!=e[st].midy) st=i;
	}
	std::cout<<ans;
	return 0;
}
```

注意事项：得开long long由于这题数据比较大，不开过不了qwq
